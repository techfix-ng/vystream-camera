cmake_minimum_required(VERSION 3.28...3.30)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)
include(compilerconfig)
include(defaults)
include(helpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
add_library(${CMAKE_PROJECT_NAME} MODULE)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
target_sources(${CMAKE_PROJECT_NAME} PRIVATE
  src/dock/vystrm-dock.cpp
  src/dock/vystrm.qrc
  src/auth/auth-manager.cpp
  src/auth/auth-manager.h
  src/auth/login-widget.cpp
  src/auth/login-widget.h
  src/qr_v4.h
)
if(WIN32)
  target_sources(${CMAKE_PROJECT_NAME} PRIVATE src/plugin.cpp)
  target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE ws2_32 shell32 advapi32 crypt32 iphlpapi gdi32 winmm)
elseif(APPLE)
  target_sources(${CMAKE_PROJECT_NAME} PRIVATE src/mac/plugin.c)
  target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE "-framework AudioToolbox" "-framework CoreAudio" "-framework Security")
else()
  message(FATAL_ERROR "VyStream OBS Dock supports Windows and macOS only")
endif()
target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE VYSTRM_QT_DOCK=1)
target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE src src/dock)
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
if(APPLE)
  # OBS's older macOS dependency bundle can still advertise the removed AGL
  # framework. Xcode 26 no longer ships it, so strip only that stale
  # transitive framework while retaining all current OBS/Qt frameworks.
  foreach(_vystrm_dependency IN ITEMS OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
    if(TARGET ${_vystrm_dependency})
      get_target_property(_vystrm_link_libraries ${_vystrm_dependency} INTERFACE_LINK_LIBRARIES)
      if(_vystrm_link_libraries)
        set(_vystrm_filtered_libraries "")
        list(LENGTH _vystrm_link_libraries _vystrm_library_count)
        math(EXPR _vystrm_last_index "${_vystrm_library_count} - 1")
        if(_vystrm_library_count GREATER 0)
          foreach(_vystrm_index RANGE 0 ${_vystrm_last_index})
            list(GET _vystrm_link_libraries ${_vystrm_index} _vystrm_item)
            if(_vystrm_item STREQUAL "-framework")
              math(EXPR _vystrm_next_index "${_vystrm_index} + 1")
              if(_vystrm_next_index LESS _vystrm_library_count)
                list(GET _vystrm_link_libraries ${_vystrm_next_index} _vystrm_next_item)
                if(_vystrm_next_item STREQUAL "AGL")
                  list(APPEND _vystrm_filtered_libraries "__VYSTRM_SKIP_AGL__")
                  continue()
                endif()
              endif()
            endif()
            if(NOT _vystrm_item STREQUAL "__VYSTRM_SKIP_AGL__" AND
               NOT _vystrm_item MATCHES "AGL(\\.framework)?$")
              list(APPEND _vystrm_filtered_libraries "${_vystrm_item}")
            endif()
          endforeach()
        endif()
        list(REMOVE_ITEM _vystrm_filtered_libraries "__VYSTRM_SKIP_AGL__")
        set_property(TARGET ${_vystrm_dependency} PROPERTY INTERFACE_LINK_LIBRARIES "${_vystrm_filtered_libraries}")
      endif()
      get_target_property(_vystrm_link_options ${_vystrm_dependency} INTERFACE_LINK_OPTIONS)
      if(_vystrm_link_options)
        list(FILTER _vystrm_link_options EXCLUDE REGEX "AGL")
        set_property(TARGET ${_vystrm_dependency} PROPERTY INTERFACE_LINK_OPTIONS "${_vystrm_link_options}")
      endif()
    endif()
  endforeach()
endif()
set_target_properties(${CMAKE_PROJECT_NAME} PROPERTIES AUTOMOC ON AUTORCC ON PREFIX "")
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME "obs-srt-camera")
